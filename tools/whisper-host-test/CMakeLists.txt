# دفتري · tools/whisper-host-test/CMakeLists.txt — host (x86_64) build of the app's whisper code for an on-computer check.
cmake_minimum_required(VERSION 3.22)
project(whisper_host_test C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(APP_CPP ${CMAKE_CURRENT_SOURCE_DIR}/../../android/app/src/main/cpp)
add_subdirectory(${APP_CPP}/whisper.cpp whisper)
add_executable(whisper_host_test main.cpp ${APP_CPP}/daftari_whisper.cpp)
target_include_directories(whisper_host_test PRIVATE ${APP_CPP})
target_link_libraries(whisper_host_test PRIVATE whisper)
